    private static function testPause():Void {
        _root.暂停=false;

        _root.暂停=true;check(PauseManager.isPausedExcept(own),"manual pause under claim blocks");
        _root.暂停=false;check(!PauseManager.isPausedExcept(own),"manual resume leaves own claim exempt");
