    public static function tick():Void {
        age++;
        if(session==null || session.failed || age>900) {check(false,"actual asset load or initializer failed:"+session.reason);finish();return;}
        if(session.pending>0 || xmlPending>0) return;
        if(phase==0) {waveAndDamage();phase=0.5;age=0;}
        else if(phase==0.5) {
            var slot:MovieClip=_root.gameworld[replacementName];
            if(age<4 || age==30) trace("[CF6_PROP_UNLOAD_WAIT] age="+age+" name="+slot._name+" depth="+slot.getDepth()+" owned="+(slot.__cf6InteractiveOwner===replacementRow.token));
            if(slot._name!=undefined && age<30) return;
            check(slot._name==undefined,"managed barrel unload completes before same-name recreation");
            if(slot._name!=undefined) {finish();return;}
            replacementMarker={};replacementForeign=_root.gameworld.createEmptyMovieClip(replacementName,_root.gameworld.getNextHighestDepth());
            replacementForeign.marker=replacementMarker;phase=0.6;age=0;
        }
        else if(phase==0.6) {
            finishReplacement();phase=1;var p:Object=spec("pod-boss","pod",1939,500);p.spawnAuthority="boss";begin("A17",[p]);
        }
        else if(phase==1) {bossPod();phase=2;p=spec("pod-timeout","pod",1939,500);p.spawnAuthority="boss";begin("A17",[p]);}
        else if(phase==2) {
            timeoutPod();phase=2.1;p=spec("pod-load-cancel","pod",1939,500);p.spawnAuthority="boss";begin("A17",[p]);
            Cf6InteractivePropService.cancelBossPods(_root.gameworld);
            check(session.bossPodsCancelled===true && session.rows[0].phase=="cancelled","boss cancellation is latched before actual resource onLoadInit");
        }
        else if(phase==2.1) {
            var cancelRow:Object=session.rows[0];
            check(cancelRow.ready && cancelRow.phase=="cancelled" && cancelRow.art._currentframe==1,"late actual load cannot reopen cancelled untriggered chamber");

            _root.gameworld.dispatcher.publish("UnitSpawn",lateChild._name);
            check(cancelRow.phase=="cancelled" && cancelRow.bornToken==undefined,"late birth event cannot revive cancelled presentation");
            phase=2.3;p=spec("pod-clear-first","pod",1939,500);p.spawnAuthority="boss";begin("A17",[p]);
        }
        else if(phase==2.3) {
            cancelRow=session.rows[0];_root.gameworld.hero._x=cancelRow.holder._x;steps(23);
